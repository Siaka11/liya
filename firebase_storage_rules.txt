// Règles de sécurité Firebase Storage pour le projet liya-a4a9f
// À copier dans la console Firebase > Storage > Rules

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // Règles temporaires pour les tests (à remplacer par des règles plus sécurisées en production)
    match /{allPaths=**} {
      allow read, write: if true;
    }
    
    // Règles spécifiques pour les dossiers de l'application
    match /restaurants/{restaurantId}/{fileName} {
      allow read: if true;
      allow write: if true;
    }
    
    match /dishes/{dishId}/{fileName} {
      allow read: if true;
      allow write: if true;
    }
    
    match /users/{userId}/{fileName} {
      allow read: if true;
      allow write: if true;
    }
    
    match /test/{fileName} {
      allow read, write: if true;
    }
  }
}
